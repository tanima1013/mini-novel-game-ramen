# ============================================================ # main.py # Google Colab用 ノベルゲーム # ============================================================ from IPython.display import display, HTML from scenario import SCENARIO, CHARACTERS import base64 import json import os import mimetypes # ============================================================ # フォルダ設定 # ============================================================ IMAGE_DIR = "/content/images" AUDIO_DIR = "/content/audio" BGM_FILE = "bgm.mp3" TYPE_SE_FILE = "type.mp3" CHOICE_SE_FILE = "choice.mp3" # ============================================================ # ファイル → Data URL # ============================================================ def file_to_data_url(path): if not os.path.exists(path): print("ファイルが見つかりません:", path) return None mime_type, _ = mimetypes.guess_type(path) if mime_type is None: mime_type = "application/octet-stream" with open(path, "rb") as f: encoded = base64.b64encode( f.read() ).decode("utf-8") return ( "data:" + mime_type + ";base64," + encoded ) # ============================================================ # 背景画像読み込み # ============================================================ BACKGROUND_DATA = {} for scene in SCENARIO: filename = scene.get("background") if not filename: continue if filename in BACKGROUND_DATA: continue path = os.path.join( IMAGE_DIR, filename ) data_url = file_to_data_url(path) if data_url: BACKGROUND_DATA[filename] = data_url print( "画像読み込み成功:", filename ) # ============================================================ # 音声読み込み # ============================================================ BGM_DATA = file_to_data_url( os.path.join( AUDIO_DIR, BGM_FILE ) ) TYPE_SE_DATA = file_to_data_url( os.path.join( AUDIO_DIR, TYPE_SE_FILE ) ) CHOICE_SE_DATA = file_to_data_url( os.path.join( AUDIO_DIR, CHOICE_SE_FILE ) ) if BGM_DATA: print("BGM読み込み成功:", BGM_FILE) if TYPE_SE_DATA: print("文字送り音読み込み成功:", TYPE_SE_FILE) if CHOICE_SE_DATA: print("選択音読み込み成功:", CHOICE_SE_FILE) # ============================================================ # JSON変換 # ============================================================ scenario_json = json.dumps( SCENARIO, ensure_ascii=False ) characters_json = json.dumps( CHARACTERS, ensure_ascii=False ) background_json = json.dumps( BACKGROUND_DATA, ensure_ascii=False ) bgm_json = json.dumps( BGM_DATA, ensure_ascii=False ) type_se_json = json.dumps( TYPE_SE_DATA, ensure_ascii=False ) choice_se_json = json.dumps( CHOICE_SE_DATA, ensure_ascii=False ) # ============================================================ # HTML # ============================================================ html = r"""
ラーメンを食べるか否か
""" # ============================================================ # データ埋め込み # ============================================================ html = html.replace( "__SCENARIO_JSON__", scenario_json ) html = html.replace( "__CHARACTERS_JSON__", characters_json ) html = html.replace( "__BACKGROUND_JSON__", background_json ) html = html.replace( "__BGM_JSON__", bgm_json ) html = html.replace( "__TYPE_SE_JSON__", type_se_json ) html = html.replace( "__CHOICE_SE_JSON__", choice_se_json ) # ============================================================ # 表示 # ============================================================ display( HTML(html) )
